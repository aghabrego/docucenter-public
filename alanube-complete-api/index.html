🌎 Servicio Alanube Completo - Dual Country PAC Integration

📋 Descripción General

El Servicio Alanube Completo en DocuCenter proporciona una implementación unificada de facturación electrónica que soporta automáticamente tanto Panamá como República Dominicana utilizando la plataforma Alanube. El sistema detecta automáticamente el país basado en la configuración PAC y aplica las validaciones, endpoints y transformaciones de datos correspondientes.

⭐ Características Principales

  • ✅ Detección Automática de País: Identifica automáticamente si usar endpoints de Panamá o República Dominicana
  • ✅ Detección Automática de Tipo de Documento: Inteligencia integrada para seleccionar el tipo correcto según datos de entrada
  • ✅ Validaciones Específicas por País: Cumplimiento con DGI Panamá y DGII República Dominicana
  • ✅ Helper de Formateo Dual: Transformación automática de datos DocuCenter según país destino
  • ✅ Procesamiento Asíncrono: Manejo de trabajos en cola con estados granulares
  • ✅ Compatibilidad Total: Sin breaking changes con implementaciones existentes
  • ✅ Testing Integral: Suite completa de pruebas para ambos países

🔍 Detección Automática de País

public function detectCountryFromPacConnection(Pacconnection $pacConnection): string
{
    $endpoint = strtolower($pacConnection->endpoint ?? '');
    $name = strtolower($pacConnection->name ?? '');

    // Detección por endpoint URL
    if (strpos($endpoint, '/pan/v1') !== false || 
        strpos($endpoint, 'panama') !== false ||
        strpos($endpoint, '.pa') !== false) {
        return 'panama';
    }

    if (strpos($endpoint, '/dom/v1') !== false || 
        strpos($endpoint, 'dominicana') !== false ||
        strpos($endpoint, '.do') !== false) {
        return 'dominican_republic';
    }

    // Detección por nombre de conexión
    if (strpos($name, 'panama') !== false || 
        strpos($name, 'pan') !== false) {
        return 'panama';
    }

    // Por defecto: República Dominicana (compatibilidad)
    return 'dominican_republic';
}

🔗 Construcción de URLs Inteligente

public function buildApiUrl(Pacconnection $pacConnection, string $endpoint): string
{
    $country = $this->detectCountryFromPacConnection($pacConnection);
    $baseUrl = rtrim($pacConnection->endpoint, '/');

    switch ($country) {
        case 'panama':
            // URLs oficiales de Panamá
            if (strpos($baseUrl, 'sandbox') !== false) {
                return "https://sandbox-api.alanube.co/pan/v1/{$endpoint}";
            } else {
                return "https://api.alanube.co/pan/v1/{$endpoint}";
            }

        case 'dominican_republic':
            // URLs oficiales de República Dominicana
            if (strpos($baseUrl, 'sandbox') !== false) {
                return "https://sandbox.alanube.co/dom/v1/{$endpoint}";
            } else {
                return "https://api.alanube.co/dom/v1/{$endpoint}";
            }

        default:
            throw new \Exception("País no soportado: {$country}");
    }
}

🤖 Detección Automática de Tipo - Panamá

private function determineDocumentTypePanama(array $data): string
{
    // 1. Tipo explícito
    if (isset($data['document_type'])) {
        return $data['document_type'];
    }

    // 2. Detección por campos específicos
    $information = $data['information'] ?? [];
    $receiver = $data['receiver'] ?? [];

    // Exportación: destinatario extranjero o destino "2"
    if (isset($information['destination']) && $information['destination'] === '2') {
        return '03'; // EXPORT
    }

    if (isset($receiver['country']) && $receiver['country'] !== 'PA') {
        return '03'; // EXPORT
    }

    // Importación: naturaleza específica
    if (isset($information['nature']) && $information['nature'] === '21') {
        return '02'; // IMPORT
    }

    // Zona franca: información presente
    if (isset($information['freeZone']) && $information['freeZone']) {
        return '08'; // FREE_ZONE
    }

    // Reembolso: naturaleza específica
    if (isset($information['nature']) && $information['nature'] === '11') {
        return '09'; // REIMBURSEMENT
    }

    // Operación extranjera: naturalezas específicas
    if (isset($information['nature']) && 
        in_array($information['nature'], ['04', '05'])) {
        return '10'; // FOREIGN_OPERATION
    }

    // Por defecto: operación interna
    return '01'; // INTERNAL_OPERATION
}

🤖 Detección Automática de Tipo - República Dominicana

private function determineDocumentTypeDominicana(array $data): int
{
    // 1. Tipo explícito
    if (isset($data['document_type'])) {
        return intval($data['document_type']);
    }

    $total = floatval(array_get($data, 'total', 0));
    $customer = array_get($data, 'customer', []);
    $customerRnc = array_get($customer, 'tax_id', '');

    // 2. RNC gubernamental (prefijos 10, 11)
    if ($this->isGovernmentalRnc($customerRnc)) {
        return 45; // GOVERNMENTAL
    }

    // 3. Campos gubernamentales
    if (isset($data['income_type']) || isset($data['transport'])) {
        return 45; // GOVERNMENTAL
    }

    // 4. Campos fiscales complejos
    if ($this->isFiscalCreditIndicator($data)) {
        return 31; // FISCAL_INVOICE
    }

    // 5. Campos de exportación
    if ($this->isExportInvoiceIndicator($data)) {
        return 46; // EXPORT_SUPPORT
    }

    // 6. Consumidor final con monto
    if (empty($customer) || $customer['id'] === 'Consumidor Final') {
        return $total > 250000 ? 31 : 32; // FISCAL vs INVOICE
    }

    // 7. RNC empresarial válido
    if ($this->hasValidBusinessRnc($customerRnc)) {
        return 31; // FISCAL_INVOICE
    }

    // 8. Por defecto
    return 32; // INVOICE
}

public static function detectCountryFromConnection(?Pacconnection $connection): string
{
    if (!$connection) {
        return 'dominican_republic'; // Por defecto
    }

    $endpoint = strtolower($connection->endpoint ?? '');
    $name = strtolower($connection->name ?? '');

    // Detección por endpoint
    if (strpos($endpoint, '/pan/v1') !== false || 
        strpos($endpoint, 'panama') !== false) {
        return 'panama';
    }

    // Detección por nombre
    if (strpos($name, 'panama') !== false || 
        strpos($name, 'pan') !== false) {
        return 'panama';
    }

    return 'dominican_republic';
}
